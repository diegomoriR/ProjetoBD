1. DESCRICAO FINAL DO SISTEMA
============================================================

O sistema gerencia eventos e os espaços usados na sua realização, 
sobre um banco PostgreSQL com 16 tabelas, 6 funções e 6 triggers. 
Uma Iniciativa agrupa as edições recorrentes (Evento); cada edição 
tem período próprio, promotores (pessoas ou organizações, vários 
por evento, cada um com um papel), atividades e inscrições. Cada 
Atividade (palestra, workshop, minicurso etc.) pode ter várias 
Sessões, realizadas em um Espaço com data, horário, limite de vagas 
e status (agendada, realizada ou cancelada). Os espaços têm capacidade 
e recursos disponíveis, e as atividades podem requerer recursos. Pessoas 
atuam nas atividades com uma função (palestrante, instrutor etc.). 
A inscrição no evento dá acesso à programação geral; sessões de atividades 
que exigem inscrição adicional têm vagas limitadas, lista de espera e 
promoção automática do primeiro da fila quando uma vaga é liberada.

Os triggers impedem sessões sobrepostas no mesmo espaço, vagas acima da 
capacidade do espaço e espaços sem os recursos requeridos pela atividade; 
controlam a confirmação, a lista de espera e a promoção nas inscrições em 
sessões; e registram em Historico_sessao o espaço e o horário anteriores 
a cada remarcação. O protótipo em Python (linha de comando) conecta ao banco, 
executa as 5 consultas definidas neste arquivo e converte perguntas em 
linguagem natural em SQL por meio de um modelo local do Ollama, exibindo o 
resultado ou a mensagem de erro.

============================================================
2. DECISOES DE PROJETO
============================================================

Q1: Promotor é uma generalização com Pessoa (CPF) e Organizacao (CNPJ) em tabelas separadas que compartilham a PK promotor_id; Evento e Promotor se relacionam em N:M via Promove (com atributo papel), permitindo vários responsáveis e promoção conjunta.
Q2: O papel não é atributo fixo da Pessoa: o papel de promoção fica em Promove.papel (por evento) e o de palestrante, instrutor ou organizador em Atua_Em.funcao (por pessoa e atividade), então uma mesma pessoa pode acumular papéis e cada papel depende do evento ou da atividade.
Q3: Atividade (título, tipo, exige_inscricao) é a oferta do programa do evento e se relaciona 1:N com Sessao, que guarda a realização concreta (espaço, início, fim, vagas, status).
Q4: Iniciativa (nome) agrupa as edições e se relaciona 1:N com Evento, que representa cada edição (edicao, data_inicio, data_fim, única por iniciativa); atividades, promotores e inscrições pertencem à edição, e só o nome da iniciativa é compartilhado.
Q5: Espaco.capacidade é o limite físico e Sessao.vagas o limite próprio da sessão (trigger garante vagas <= capacidade); só atividades com exige_inscricao aceitam inscrição em Inscricao_sessao, e um trigger confirma a inscrição se houver vaga ou a coloca em lista de espera (posicao_espera), promovendo o primeiro da fila quando uma vaga é liberada.
Q6: Sessao guarda apenas a programação atual e Historico_sessao registra espaço, início e fim anteriores com alterado_em, gravados por trigger a cada mudança, o que permite recuperar o histórico; outros triggers impedem conflito de espaço e horário e exigem que o espaço tenha os recursos da atividade.

-- T1: impede duas sessões não canceladas sobrepostas no mesmo espaço
CREATE FUNCTION fn_sessao_conflito_espaco() RETURNS trigger AS $$
BEGIN
    IF NEW.status = 'cancelada' THEN
        RETURN NEW;
    END IF;

    IF EXISTS (
        SELECT 1
        FROM Sessao s
        WHERE s.espaco_id = NEW.espaco_id
          AND s.status <> 'cancelada'
          AND s.inicio < NEW.fim
          AND s.fim > NEW.inicio
          AND (TG_OP = 'INSERT' OR s.sessao_id <> NEW.sessao_id)
    ) THEN
        RAISE EXCEPTION 'Conflito: o espaco % ja possui sessao no intervalo % a %',
            NEW.espaco_id, NEW.inicio, NEW.fim;
    END IF;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_sessao_conflito_espaco
BEFORE INSERT OR UPDATE ON Sessao
FOR EACH ROW EXECUTE FUNCTION fn_sessao_conflito_espaco();

-- T2: o limite de vagas da sessão não pode exceder a capacidade física do espaço
CREATE FUNCTION fn_sessao_vagas_capacidade() RETURNS trigger AS $$
DECLARE
    v_capacidade integer;
BEGIN
    IF NEW.vagas IS NOT NULL THEN
        SELECT capacidade INTO v_capacidade
        FROM Espaco
        WHERE espaco_id = NEW.espaco_id;

        IF NEW.vagas > v_capacidade THEN
            RAISE EXCEPTION 'Vagas (%) excedem a capacidade do espaco (%)',
                NEW.vagas, v_capacidade;
        END IF;
    END IF;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_sessao_vagas_capacidade
BEFORE INSERT OR UPDATE ON Sessao
FOR EACH ROW EXECUTE FUNCTION fn_sessao_vagas_capacidade();

-- T3: o espaço deve dispor de todos os recursos requeridos pela atividade
CREATE FUNCTION fn_sessao_recursos() RETURNS trigger AS $$
BEGIN
    IF NEW.status = 'cancelada' THEN
        RETURN NEW;
    END IF;

    IF EXISTS (
        SELECT 1
        FROM Requer r
        WHERE r.atividade_id = NEW.atividade_id
          AND NOT EXISTS (
              SELECT 1
              FROM Dispoe d
              WHERE d.espaco_id = NEW.espaco_id
                AND d.recurso_id = r.recurso_id
          )
    ) THEN
        RAISE EXCEPTION 'O espaco % nao dispoe de todos os recursos requeridos pela atividade %',
            NEW.espaco_id, NEW.atividade_id;
    END IF;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_sessao_recursos
BEFORE INSERT OR UPDATE ON Sessao
FOR EACH ROW EXECUTE FUNCTION fn_sessao_recursos();

-- T4: registra em Historico_sessao a programação anterior quando espaço ou horário mudam
CREATE FUNCTION fn_sessao_historico() RETURNS trigger AS $$
BEGIN
    INSERT INTO Historico_sessao (sessao_id, espaco_ant, inicio_ant, fim_ant)
    VALUES (OLD.sessao_id, OLD.espaco_id, OLD.inicio, OLD.fim);

    RETURN NULL;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_sessao_historico
AFTER UPDATE ON Sessao
FOR EACH ROW
WHEN (OLD.espaco_id IS DISTINCT FROM NEW.espaco_id
      OR OLD.inicio IS DISTINCT FROM NEW.inicio
      OR OLD.fim IS DISTINCT FROM NEW.fim)
EXECUTE FUNCTION fn_sessao_historico();

-- T5: valida a inscrição em sessão e decide entre vaga confirmada e lista de espera
CREATE FUNCTION fn_inscricao_sessao_vaga() RETURNS trigger AS $$
DECLARE
    v_sessao_status varchar;
    v_vagas integer;
    v_atividade integer;
    v_exige boolean;
    v_evento_atividade integer;
    v_evento_inscricao integer;
    v_status_inscricao varchar;
    v_ocupadas integer;
    v_posicao integer;
BEGIN
    -- trava a sessão para serializar inscrições concorrentes
    SELECT status, vagas, atividade_id
      INTO v_sessao_status, v_vagas, v_atividade
    FROM Sessao
    WHERE sessao_id = NEW.sessao_id
    FOR UPDATE;

    IF NOT FOUND THEN
        RAISE EXCEPTION 'Sessao % inexistente', NEW.sessao_id;
    END IF;

    IF v_sessao_status = 'cancelada' THEN
        RAISE EXCEPTION 'Sessao % cancelada: inscricao nao permitida', NEW.sessao_id;
    END IF;

    SELECT exige_inscricao, evento_id
      INTO v_exige, v_evento_atividade
    FROM Atividade
    WHERE atividade_id = v_atividade;

    IF NOT v_exige THEN
        RAISE EXCEPTION 'A atividade da sessao % nao exige inscricao adicional', NEW.sessao_id;
    END IF;

    SELECT evento_id, status
      INTO v_evento_inscricao, v_status_inscricao
    FROM Inscricao
    WHERE inscricao_id = NEW.inscricao_id;

    IF NOT FOUND THEN
        RAISE EXCEPTION 'Inscricao % inexistente', NEW.inscricao_id;
    END IF;

    IF v_status_inscricao = 'cancelada' THEN
        RAISE EXCEPTION 'Inscricao % cancelada no evento', NEW.inscricao_id;
    END IF;

    IF v_evento_inscricao <> v_evento_atividade THEN
        RAISE EXCEPTION 'Inscricao % pertence a outro evento', NEW.inscricao_id;
    END IF;

    SELECT COUNT(*) INTO v_ocupadas
    FROM Inscricao_sessao
    WHERE sessao_id = NEW.sessao_id
      AND status = 'confirmada';

    IF v_vagas IS NULL OR v_ocupadas < v_vagas THEN
        NEW.status := 'confirmada';
        NEW.posicao_espera := NULL;
    ELSE
        SELECT COALESCE(MAX(posicao_espera), 0) + 1 INTO v_posicao
        FROM Inscricao_sessao
        WHERE sessao_id = NEW.sessao_id
          AND status = 'lista_espera';

        NEW.status := 'lista_espera';
        NEW.posicao_espera := v_posicao;
    END IF;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_inscricao_sessao_vaga
BEFORE INSERT ON Inscricao_sessao
FOR EACH ROW EXECUTE FUNCTION fn_inscricao_sessao_vaga();

-- T6: ao liberar uma vaga confirmada, promove o primeiro da lista de espera
CREATE FUNCTION fn_promove_lista_espera() RETURNS trigger AS $$
DECLARE
    v_status_sessao varchar;
    v_proximo integer;
BEGIN
    IF OLD.status <> 'confirmada' THEN
        RETURN NULL;
    END IF;

    IF TG_OP = 'UPDATE' THEN
        IF NEW.status = 'confirmada' THEN
            RETURN NULL;
        END IF;
    END IF;

    SELECT status INTO v_status_sessao
    FROM Sessao
    WHERE sessao_id = OLD.sessao_id;

    -- sessão removida (exclusão em cascata) ou cancelada: nada a promover
    IF NOT FOUND OR v_status_sessao = 'cancelada' THEN
        RETURN NULL;
    END IF;

    SELECT inscricao_id INTO v_proximo
    FROM Inscricao_sessao
    WHERE sessao_id = OLD.sessao_id
      AND status = 'lista_espera'
    ORDER BY posicao_espera
    LIMIT 1
    FOR UPDATE;

    IF FOUND THEN
        UPDATE Inscricao_sessao
        SET status = 'confirmada', posicao_espera = NULL
        WHERE inscricao_id = v_proximo
          AND sessao_id = OLD.sessao_id;

        UPDATE Inscricao_sessao
        SET posicao_espera = posicao_espera - 1
        WHERE sessao_id = OLD.sessao_id
          AND status = 'lista_espera';
    END IF;

    RETURN NULL;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_promove_lista_espera
AFTER UPDATE OR DELETE ON Inscricao_sessao
FOR EACH ROW EXECUTE FUNCTION fn_promove_lista_espera();

CONSULTA 1
OBJETIVO: Listar as atividades da SECOMP 2026 com tipo, se exigem inscrição, quantidade de sessões não canceladas e período em que acontecem, incluindo as que ainda não têm sessão.
SQL:
SELECT a.atividade_id,
       a.titulo                          AS atividade,
       a.tipo,
       a.exige_inscricao,
       COUNT(s.sessao_id)                AS qtd_sessoes,
       MIN(s.inicio)                     AS primeira_sessao,
       MAX(s.fim)                        AS fim_da_ultima_sessao
FROM Atividade a
JOIN Evento     ev ON ev.evento_id    = a.evento_id
JOIN Iniciativa i  ON i.iniciativa_id = ev.iniciativa_id
LEFT JOIN Sessao s ON s.atividade_id = a.atividade_id
                  AND s.status <> 'cancelada'
WHERE i.nome = 'Semana de Computação (SECOMP)'
  AND ev.edicao = '2026'
GROUP BY a.atividade_id, a.titulo, a.tipo, a.exige_inscricao
ORDER BY MIN(s.inicio) NULLS LAST, a.titulo;

CONSULTA 2
OBJETIVO: Identificar as sessões agendadas com vagas limitadas que estão com 80% ou mais das vagas confirmadas, mostrando quantos aguardam na lista de espera.
SQL:
SELECT s.sessao_id,
       a.titulo                                              AS atividade,
       esp.nome                                              AS espaco,
       esp.capacidade,
       s.vagas,
       COUNT(isx.inscricao_id) FILTER (WHERE isx.status = 'confirmada')   AS confirmadas,
       COUNT(isx.inscricao_id) FILTER (WHERE isx.status = 'lista_espera') AS em_espera,
       ROUND(100.0 * COUNT(isx.inscricao_id) FILTER (WHERE isx.status = 'confirmada')
             / s.vagas, 1)                                   AS ocupacao_pct
FROM Sessao s
JOIN Atividade a   ON a.atividade_id = s.atividade_id
JOIN Espaco    esp ON esp.espaco_id  = s.espaco_id
LEFT JOIN Inscricao_sessao isx ON isx.sessao_id = s.sessao_id
WHERE s.vagas IS NOT NULL
  AND s.status = 'agendada'
GROUP BY s.sessao_id, a.titulo, esp.nome, esp.capacidade, s.vagas
HAVING COUNT(isx.inscricao_id) FILTER (WHERE isx.status = 'confirmada') >= 0.8 * s.vagas
ORDER BY ocupacao_pct DESC, em_espera DESC;

CONSULTA 3
OBJETIVO: Descobrir quais espaços foram usados acima da média de horas de sessões não canceladas, contando também os espaços sem uso no cálculo da média.
SQL:
WITH uso AS (
    SELECT esp.espaco_id,
           esp.nome,
           esp.capacidade,
           COUNT(s.sessao_id)                                              AS qtd_sessoes,
           COALESCE(SUM(EXTRACT(EPOCH FROM (s.fim - s.inicio)) / 3600), 0) AS horas
    FROM Espaco esp
    LEFT JOIN Sessao s ON s.espaco_id = esp.espaco_id
                      AND s.status <> 'cancelada'
    GROUP BY esp.espaco_id, esp.nome, esp.capacidade
)
SELECT nome AS espaco,
       capacidade,
       qtd_sessoes,
       ROUND(horas, 1)                         AS horas_de_uso,
       ROUND((SELECT AVG(horas) FROM uso), 1)  AS media_geral
FROM uso
WHERE horas > (SELECT AVG(horas) FROM uso)
ORDER BY horas DESC;

CONSULTA 4
OBJETIVO: Mostrar as sessões que tiveram espaço ou horário alterados, comparando a programação anterior (registrada no histórico) com a programação atual.
SQL:
SELECT s.sessao_id,
       a.titulo        AS atividade,
       h.alterado_em,
       ea.nome         AS espaco_anterior,
       h.inicio_ant,
       h.fim_ant,
       ec.nome         AS espaco_atual,
       s.inicio        AS inicio_atual,
       s.fim           AS fim_atual
FROM Historico_sessao h
JOIN Sessao    s  ON s.sessao_id    = h.sessao_id
JOIN Atividade a  ON a.atividade_id = s.atividade_id
JOIN Espaco    ea ON ea.espaco_id   = h.espaco_ant     -- espaço anterior
JOIN Espaco    ec ON ec.espaco_id   = s.espaco_id      -- espaço atual
ORDER BY h.alterado_em DESC, s.sessao_id;

CONSULTA 5
OBJETIVO: Para as sessões lotadas com lista de espera, listar em quais outros espaços (livres no mesmo horário e com todos os recursos requeridos) elas caberiam mais gente e quais inscrições da fila seriam atendidas.
SQL:
SELECT s.sessao_id,
       a.titulo                           AS atividade,
       ea.nome                            AS espaco_atual,
       s.vagas                            AS vagas_atuais,
       en.nome                            AS espaco_alternativo,
       en.capacidade                      AS capacidade_alternativa,
       en.capacidade - s.vagas            AS vagas_adicionais,
       w.inscricao_id                     AS inscricao_atendida,
       w.posicao_espera
FROM Sessao s
JOIN Atividade a  ON a.atividade_id = s.atividade_id
JOIN Espaco    ea ON ea.espaco_id   = s.espaco_id
JOIN Espaco    en ON en.espaco_id  <> s.espaco_id
                 AND en.capacidade  > s.vagas
JOIN Inscricao_sessao w ON w.sessao_id = s.sessao_id
                       AND w.status = 'lista_espera'
                       AND w.posicao_espera <= en.capacidade - s.vagas
WHERE s.status = 'agendada'
  AND s.vagas IS NOT NULL
  AND s.vagas <= (SELECT COUNT(*)                    -- sessão lotada
                  FROM Inscricao_sessao c
                  WHERE c.sessao_id = s.sessao_id
                    AND c.status = 'confirmada')
  AND NOT EXISTS (                                   -- o espaço alternativo está livre no horário
        SELECT 1
        FROM Sessao o
        WHERE o.espaco_id = en.espaco_id
          AND o.status <> 'cancelada'
          AND o.inicio < s.fim
          AND o.fim    > s.inicio)
  AND NOT EXISTS (                                   -- e dispõe de todos os recursos requeridos
        SELECT 1
        FROM Requer r
        WHERE r.atividade_id = s.atividade_id
          AND NOT EXISTS (
                SELECT 1
                FROM Dispoe d
                WHERE d.espaco_id  = en.espaco_id
                  AND d.recurso_id = r.recurso_id))
ORDER BY s.sessao_id, en.capacidade DESC, w.posicao_espera;
